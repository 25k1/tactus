# Голосовые команды Tactus.
# Формат: команда = фраза | фраза
# (а|б) — одно из слов, [слово] — необязательно, {number} — число, {name} — название.
# «=> 10» в конце фразы задаёт число, если его не назвали.

play = играй | играть | продолжи | продолжай | продолжить | включи | включи музыку | воспроизведи | плей
pause = пауза | (поставь|поставить) на паузу | стоп | замолчи | хватит | остановись
stop = останови (воспроизведение|музыку) | выключи музыку
next = (следующий|следующая|следующую|следующее) [трек|песня|песню|композиция|композицию] | дальше | пропусти [трек|песню] | (переключи|переключай) [трек|песню] | (включи|поставь) (следующий|следующую) [трек|песню] | некст
previous = (предыдущий|предыдущая|предыдущую|предыдущее) [трек|песня|песню|композиция|композицию] | назад | (верни|вернуть) [предыдущий|предыдущую] [трек|песню] | (прошлый|прошлую) (трек|песню) | (включи|поставь) (предыдущий|предыдущую) [трек|песню]

volume-up = громче | погромче | еще громче | (сделай|сделать) громче | (прибавь|добавь|увеличь) [громкость|звук]
volume-down = тише | потише | еще тише | (сделай|сделать) тише | (убавь|уменьши) [громкость|звук]
volume = громкость [на] {number} [процентов|процента|процент] | (сделай|поставь|установи) громкость [на] {number} [процентов|процента|процент]
faster = быстрее | побыстрее | (ускорь|ускорить) [воспроизведение]
slower = медленнее | помедленнее | (замедли|замедлить) [воспроизведение]
normal-speed = (обычная|нормальная|обычную|нормальную) скорость | (верни|сбрось) скорость
forward = (вперед|перемотай вперед|промотай вперед|перемотай|промотай) [на] {number} (секунд|секунды|секунду) | (перемотай|промотай) вперед => 10 | вперед на (минуту|минутку) => 60
rewind = (назад|перемотай назад|отмотай назад|отмотай) [на] {number} (секунд|секунды|секунду) | (перемотай|отмотай) назад => 10 | отмотай => 10 | назад на (минуту|минутку) => 60

repeat-off = (выключи|отключи|убери) (повтор|повторение) | без повтора | не повторяй
repeat-one = (повторяй|повтори) (этот трек|эту песню|трек|песню) | повтор (трека|песни|одного)
repeat-all = (повторяй|повтори) (все|всё|очередь|список) | повтор (всех|очереди|всего)
shuffle-on = (перемешай|перемешать) [очередь|треки|все] | (включи|включить) (перемешивание|случайный порядок) | в случайном порядке
shuffle-off = (выключи|отключи) (перемешивание|случайный порядок) | по порядку | не перемешивай

sleep-minutes = (выключись|выключи музыку|останови музыку|усни|засыпай) через {number} [минут|минуты|минуту] | таймер сна [на] {number} [минут|минуты|минуту]
sleep-end-of-track = (выключись|выключи музыку|усни|засыпай) после (этого трека|этой песни|трека|песни) | после этой песни (выключись|усни)
sleep-off = (выключи|отключи|отмени) таймер [сна] | не (выключайся|засыпай)
whats-playing = что (играет|звучит|это за песня|это за трек) | какой это трек | какая это песня | кто (поет|исполняет|это) | что сейчас играет

play-artist = (включи|поставь|запусти|сыграй|играй|давай) (исполнителя|артиста|группу|певца|певицу) {name} | (включи|поставь|запусти) (песни|музыку|треки) (исполнителя|группы|артиста) {name}
play-album = (включи|поставь|запусти|сыграй|играй|давай) альбом {name}
play-playlist = (включи|поставь|запусти|сыграй|играй|давай) (плейлист|подборку|список) {name}
play-track = (включи|поставь|запусти|сыграй|играй|давай) (песню|трек|композицию) {name}
play-name = (включи|поставь|запусти|сыграй|играй|давай) {name} | хочу (послушать|слушать) {name}
play-likes = (включи|поставь|запусти|сыграй) (мои лайки|любимое|любимые [треки|песни]|мне нравится|понравившиеся [треки|песни]) | (мои|моих) (лайки|любимых)
play-wave = (включи|поставь|запусти) (мою волну|волну|радио) | моя волна | (включи|поставь) (что нибудь|что то|музыку) на мой вкус
play-calm = (включи|поставь|запусти) (что нибудь|что то|музыку) (спокойное|спокойную|для релакса|расслабляющее) | (включи|поставь) спокойную музыку
play-energetic = (включи|поставь|запусти) (что нибудь|что то|музыку) (бодрое|бодрую|энергичное|энергичную) | (включи|поставь) бодрую музыку
play-fun = (включи|поставь|запусти) (что нибудь|что то|музыку) (веселое|веселую) | (включи|поставь) веселую музыку
play-sad = (включи|поставь|запусти) (что нибудь|что то|музыку) (грустное|грустную) | (включи|поставь) грустную музыку

like = (мне нравится|нравится|лайк|лайкни|поставь лайк|в избранное) [этот трек|эту песню] | отличная песня | (добавь|сохрани) в (любимое|избранное|лайки)
unlike = (убери|сними|удали) (лайк|из любимого|из избранного|из лайков)
dislike = (не нравится|мне не нравится|дизлайк|не люблю) [эту песню|этот трек] | (больше не (ставь|включай)) [эту песню|этот трек]
add-to-playlist = (добавь|добавить|сохрани) [этот трек|эту песню] в (плейлист|подборку) {name}
download = (скачай|загрузи|сохрани на компьютер) [этот трек|эту песню]
go-to-album = (открой|покажи|перейди к|перейди в) альбом | какой это альбом
go-to-artist = (открой|покажи|перейди к) (исполнителя|артиста|группу)

open-search = (открой|покажи) поиск | к поиску
open-wave = (открой|покажи) (мою волну|волну)
open-for-you = (открой|покажи) (рекомендации|для вас|для тебя|подборки)
open-collection = (открой|покажи) (коллекцию|мою коллекцию|медиатеку)
open-playlists = (открой|покажи) (плейлисты|мои плейлисты)
open-radio = (открой|покажи) (радио|станции|радиостанции)
open-podcasts = (открой|покажи) (подкасты|книги|аудиокниги|подкасты и книги)
open-history = (открой|покажи) историю | что я слушал
open-queue = (открой|покажи) очередь | что дальше | что будет дальше
open-lyrics = (открой|покажи) (текст|слова) [песни] | слова песни
open-downloads = (открой|покажи) (загрузки|скачанное|скачанные)
search = (найди|поищи|искать|ищи|поиск) {name}
read-selection = (что выбрано|прочитай выбранное|что под курсором)
go-back = вернись [назад] | (на|к) (предыдущей странице|предыдущую страницу) | обратно
help = (что можно сказать|что ты умеешь|какие есть команды|помощь|справка) | список команд
yes = да | давай | конечно | ага | угу | верно | правильно
no = нет | не надо | отмена | отмени | не то
